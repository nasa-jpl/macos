cmake_minimum_required(VERSION 3.21)
project(giza C Fortran)

# ==============================================================================
# Version
# ==============================================================================
set(GIZA_VERSION_MAJOR 1)
set(GIZA_VERSION_MINOR 2)
set(GIZA_VERSION_MICRO 0)
set(VERSION "${GIZA_VERSION_MAJOR}.${GIZA_VERSION_MINOR}.${GIZA_VERSION_MICRO}")

# ==============================================================================
# X11 — required for interactive window on Linux/macOS
# ==============================================================================
if(NOT WIN32)
  find_package(X11 REQUIRED)
endif()

# ==============================================================================
# zlib and libpng — built from bundled sources via add_subdirectory
# ==============================================================================

# zlib
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS   OFF CACHE BOOL "" FORCE)
add_subdirectory(subprojects/zlib)

# zlib unconditionally creates both shared and static targets.
# Exclude the shared library from the build to avoid conflicts.
if(TARGET zlib)
  set_target_properties(zlib PROPERTIES EXCLUDE_FROM_ALL TRUE)
endif()

# Ensure zlibstatic target exists — zlib CMakeLists names it differently on some platforms
if(NOT TARGET zlibstatic AND TARGET zlib)
  add_library(zlibstatic ALIAS zlib)
endif()

set(zlib_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/subprojects/zlib)
set(zlib_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/subprojects/zlib)

# libpng — point it to the bundled zlib
# The zlibstatic target output name differs by platform:
#   Windows: zlibstatic.lib  (zlib CMakeLists uses OUTPUT_NAME zlibstatic on MSVC)
#   Linux:   libz.a          (zlib CMakeLists uses OUTPUT_NAME z on non-MSVC)
if(WIN32)
  set(_zlib_filename "${CMAKE_STATIC_LIBRARY_PREFIX}zlibstatic${CMAKE_STATIC_LIBRARY_SUFFIX}")
else()
  set(_zlib_filename "${CMAKE_STATIC_LIBRARY_PREFIX}z${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
set(ZLIB_LIBRARY
    ${CMAKE_BINARY_DIR}/lib/${_zlib_filename}
    CACHE FILEPATH "" FORCE)
set(ZLIB_INCLUDE_DIR
    ${CMAKE_CURRENT_SOURCE_DIR}/subprojects/zlib
    ${CMAKE_CURRENT_BINARY_DIR}/subprojects/zlib
    CACHE STRING "" FORCE)
set(ZLIB_FOUND TRUE CACHE BOOL "" FORCE)
set(PNG_SHARED     OFF CACHE BOOL "" FORCE)
set(PNG_STATIC     ON  CACHE BOOL "" FORCE)
set(PNG_TESTS      OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS      OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory(subprojects/libpng)

# Explicitly wire png_static to zlibstatic so linker gets the right lib
target_link_libraries(png_static PRIVATE zlibstatic)

set(libpng_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/subprojects/libpng)
set(libpng_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/subprojects/libpng)

# ==============================================================================
# Subprojects — pixman first, then cairo (depends on pixman, zlib, libpng)
# ==============================================================================
add_subdirectory(subprojects/pixman)
add_subdirectory(subprojects/cairo)

# ==============================================================================
# Create cairo/ forwarding headers
# giza includes cairo headers as <cairo/cairo.h>, <cairo/cairo-ps.h> etc.
# but cairo exposes them directly in src/. Create forwarding headers.
# ==============================================================================
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/cairo)
foreach(_hdr
    cairo.h
    cairo-deprecated.h
    cairo-ps.h
    cairo-pdf.h
    cairo-svg.h
    cairo-script.h
    cairo-ft.h
    cairo-win32.h
    cairo-xlib.h
    cairo-tee.h
    cairo-version.h
)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/cairo/${_hdr}
"/* Forwarding header - generated by CMake */
#include \"${CMAKE_CURRENT_SOURCE_DIR}/subprojects/cairo/src/${_hdr}\"
"
    )
endforeach()

# ==============================================================================
# Windows compatibility — create unistd.h stub
# giza sources use several POSIX functions not available on Windows
# ==============================================================================
if(WIN32)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/unistd.h
"/* unistd.h stub for Windows - generated by CMake */
#pragma once
#include <io.h>
#include <process.h>
#include <direct.h>
#include <string.h>
#include <windows.h>
#define access   _access
#define getcwd   _getcwd
#define getpid   _getpid
#define sleep(x) Sleep((x)*1000)
#define usleep(x) Sleep((x)/1000)

/* strcasecmp / strncasecmp */
#ifndef strcasecmp
#  define strcasecmp  _stricmp
#endif
#ifndef strncasecmp
#  define strncasecmp _strnicmp
#endif

/* getlogin stub */
static inline char* getlogin(void) { return \"user\"; }

/* strsep implementation */
static inline char* strsep(char** stringp, const char* delim) {
    char* start = *stringp;
    char* p;
    if (!start) return NULL;
    p = start + strcspn(start, delim);
    if (*p) { *p = '\\0'; *stringp = p + 1; }
    else     { *stringp = NULL; }
    return start;
}
"
  )
endif()

#endif /* GIZA_CONFIG_H */

# ==============================================================================
# Platform-specific driver selection
# On Windows: giza-driver-win.c  (Win32 HWND via separate thread)
# On Linux:   giza-driver-xw.c   (X11/Xlib)
# ==============================================================================
if(WIN32)
  set(GIZA_PLATFORM_DRIVER giza-driver-win.c)
  set(GIZA_PLATFORM_DEFINES _GIZA_HAS_WIN)
  set(GIZA_PLATFORM_LIBS "")
  set(GIZA_PLATFORM_INCLUDES "")
else()
  set(GIZA_PLATFORM_DRIVER giza-driver-xw.c)
  set(GIZA_PLATFORM_DEFINES _GIZA_HAS_XW)
  set(GIZA_PLATFORM_LIBS ${X11_LIBRARIES})
  set(GIZA_PLATFORM_INCLUDES ${X11_INCLUDE_DIR})
endif()

    HAVE_CONFIG_H
    _CRT_SECURE_NO_WARNINGS
    _USE_MATH_DEFINES
    ${GIZA_PLATFORM_DEFINES}
    $<$<BOOL:${WIN32}>:WIN32 _WINDOWS _CRT_NONSTDC_NO_DEPRECATE strcasecmp=_stricmp strncasecmp=_strnicmp CAIRO_WIN32_STATIC_BUILD>

target_link_libraries(giza PUBLIC cairo png_static ${GIZA_PLATFORM_LIBS})
