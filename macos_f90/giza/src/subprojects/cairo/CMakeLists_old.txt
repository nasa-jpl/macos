# ==============================================================================

#define CAIRO_CONFIG_H

#define PACKAGE         \"cairo\"
#define PACKAGE_VERSION \"1.13.1\"
#define PACKAGE_BUGREPORT \"\"
#define PACKAGE_NAME    \"cairo\"

#define PACKAGE_TARNAME \"cairo\"
#define PACKAGE_URL     \"\"

#define HAVE_STDINT_H   1
#define HAVE_STDLIB_H   1
#define HAVE_STRING_H   1
#define HAVE_INTTYPES_H 1
#define HAVE_MEMORY_H   1

file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config.h     ${CAIRO_CONFIG_H_CONTENT})

# ==============================================================================
# Target
# ==============================================================================
add_library(cairo STATIC ${CAIRO_CORE_SOURCES})

target_compile_definitions(cairo PUBLIC
    $<$<BOOL:${WIN32}>:CAIRO_WIN32_STATIC_BUILD>
)

target_compile_definitions(cairo PRIVATE
    HAVE_CONFIG_H
    _USE_MATH_DEFINES
    _CRT_SECURE_NO_WARNINGS
    $<$<BOOL:${WIN32}>:WIN32 _WINDOWS CAIRO_WIN32_STATIC_BUILD>
    CAIRO_HAS_IMAGE_SURFACE=1
    CAIRO_HAS_WIN32_SURFACE=1
    CAIRO_HAS_WIN32_FONT=1
    CAIRO_HAS_SVG_SURFACE=1
    CAIRO_HAS_TEE_SURFACE=1
    CAIRO_HAS_SCRIPT_SURFACE=1
    CAIRO_HAS_PIXMAN_RENDERER=1
    CAIRO_HAS_PDF_SURFACE=1
    CAIRO_HAS_PS_SURFACE=1
    CAIRO_HAS_PNG_FUNCTIONS=1
)

target_include_directories(cairo
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/src
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_BINARY_DIR}
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/win32
        ${CMAKE_CURRENT_BINARY_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../zlib
        ${CMAKE_CURRENT_BINARY_DIR}/../zlib
        ${CMAKE_CURRENT_SOURCE_DIR}/../libpng
        ${CMAKE_CURRENT_BINARY_DIR}/../libpng
)

target_link_libraries(cairo PUBLIC pixman zlibstatic png_static gdi32 msimg32 user32)
