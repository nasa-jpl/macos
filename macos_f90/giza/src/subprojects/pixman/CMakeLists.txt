cmake_minimum_required(VERSION 3.21)
project(pixman C)

# ==============================================================================
# Version
# ==============================================================================
set(PIXMAN_VERSION_MAJOR 0)
set(PIXMAN_VERSION_MINOR 35)
set(PIXMAN_VERSION_MICRO 1)

# ==============================================================================
# Generate pixman-version.h
# ==============================================================================
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/pixman/pixman-version.h.in
    ${CMAKE_CURRENT_BINARY_DIR}/pixman/pixman-version.h
    @ONLY
)

# ==============================================================================
# Architecture detection — x86-64 gets SSE2/SSSE3; arm64 (Apple Silicon) uses
# pixman's generic C paths (the x86 SIMD sources contain x86 inline asm that
# clang rejects on arm64).  pixman-x86.c is fully #if-guarded on the USE_SSE*
# macros, so leaving them undefined compiles it to an empty TU.
# ==============================================================================
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64|i[3-6]86")
    set(PIXMAN_X86 TRUE)
else()
    set(PIXMAN_X86 FALSE)   # arm64 / aarch64 / other — generic C only
endif()

# ==============================================================================
# Generate config.h
# Autoconf normally generates this — we provide a minimal hand-written version.
# ==============================================================================
if(PIXMAN_X86)
    set(_PIXMAN_SIMD_DEFS
"/* Enable x86 SIMD optimizations — MMX disabled on 64-bit */
#define USE_SSE2      1
#define USE_SSSE3     1")
else()
    set(_PIXMAN_SIMD_DEFS
"/* arm64/other: no x86 SIMD — generic C paths (correct, just unaccelerated) */
#undef USE_SSE2
#undef USE_SSSE3")
endif()

file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config.h
"/* config.h for pixman — generated by CMake (arch-aware) */
#ifndef PIXMAN_CONFIG_H
#define PIXMAN_CONFIG_H

#define PACKAGE \"pixman\"
#define PACKAGE_VERSION \"0.35.1\"

/* intrin.h provides __cpuid on MSVC/Intel Windows */
#if defined(_MSC_VER) && defined(_WIN32)
#  include <intrin.h>
#endif

/* Compiler supports __builtin_clz */
#ifdef _MSC_VER
#  define HAVE_BUILTIN_CLZ 0
#else
#  define HAVE_BUILTIN_CLZ 1
#endif

${_PIXMAN_SIMD_DEFS}

/* Disable unsupported platforms (ARM NEON left off — generic C is enough for
   MACOS's occasional CLI plotting; enable later if profiling warrants). */
#undef USE_X86_MMX
#undef USE_ARM_NEON
#undef USE_ARM_SIMD
#undef USE_MIPS_DSPR2
#undef USE_VMX
#undef USE_LOONGSON_MMI

/* TLS support */
#ifdef _MSC_VER
#  define PIXMAN_USE_PRAGMA_TLS 1
#else
#  define HAVE_PTHREADS 1
#endif

#endif /* PIXMAN_CONFIG_H */
"
)

# ==============================================================================
# Core sources (from Makefile.sources)
# ==============================================================================
set(PIXMAN_CORE_SOURCES
    pixman/pixman.c
    pixman/pixman-access.c
    pixman/pixman-access-accessors.c
    pixman/pixman-bits-image.c
    pixman/pixman-combine32.c
    pixman/pixman-combine-float.c
    pixman/pixman-conical-gradient.c
    pixman/pixman-filter.c
    pixman/pixman-x86.c
    pixman/pixman-mips.c
    pixman/pixman-arm.c
    pixman/pixman-ppc.c
    pixman/pixman-edge.c
    pixman/pixman-edge-accessors.c
    pixman/pixman-fast-path.c
    pixman/pixman-glyph.c
    pixman/pixman-general.c
    pixman/pixman-gradient-walker.c
    pixman/pixman-image.c
    pixman/pixman-implementation.c
    pixman/pixman-linear-gradient.c
    pixman/pixman-matrix.c
    pixman/pixman-noop.c
    pixman/pixman-radial-gradient.c
    pixman/pixman-region16.c
    pixman/pixman-region32.c
    pixman/pixman-solid-fill.c
    pixman/pixman-timer.c
    pixman/pixman-trap.c
    pixman/pixman-utils.c
)

# ==============================================================================
# x86 SIMD sources
# Note: MMX is disabled on x86-64 with Intel icx (not supported in 64-bit mode)
# SSE2 is always enabled on x86-64 — no arch flag needed
# SSSE3 requires /arch:AVX or higher with icx on 64-bit
# ==============================================================================
# x86 SIMD translation units — only compiled on x86 (they carry x86 asm/intrinsics)
if(PIXMAN_X86)
    set(PIXMAN_SSE2_SOURCES  pixman/pixman-sse2.c)
    set(PIXMAN_SSSE3_SOURCES pixman/pixman-ssse3.c)
else()
    set(PIXMAN_SSE2_SOURCES  "")
    set(PIXMAN_SSSE3_SOURCES "")
endif()

# ==============================================================================
# Target
# ==============================================================================
add_library(pixman STATIC
    ${PIXMAN_CORE_SOURCES}
    ${PIXMAN_SSE2_SOURCES}
    ${PIXMAN_SSSE3_SOURCES}
)

target_compile_definitions(pixman PRIVATE
    HAVE_CONFIG_H
    _USE_MATH_DEFINES
    $<$<BOOL:${WIN32}>:WIN32 _WINDOWS>
)
if(PIXMAN_X86)
    target_compile_definitions(pixman PRIVATE USE_SSE2 USE_SSSE3)
    # SSE2 is implicit on x86-64; SSSE3 needs AVX flag with icx 64-bit (Windows)
    # On Linux with icx/gcc, use -msse2 / -mssse3
    if(WIN32 AND (MSVC OR CMAKE_C_COMPILER_ID MATCHES "IntelLLVM"))
        set_source_files_properties(pixman/pixman-ssse3.c PROPERTIES COMPILE_FLAGS "/arch:AVX")
    else()
        set_source_files_properties(pixman/pixman-sse2.c  PROPERTIES COMPILE_FLAGS "-msse2")
        set_source_files_properties(pixman/pixman-ssse3.c PROPERTIES COMPILE_FLAGS "-mssse3")
    endif()
endif()

target_include_directories(pixman
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/pixman   # pixman.h
        ${CMAKE_CURRENT_BINARY_DIR}/pixman   # pixman-version.h
    PRIVATE
        ${CMAKE_CURRENT_BINARY_DIR}          # config.h
)
